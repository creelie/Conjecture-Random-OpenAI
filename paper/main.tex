\title{Elementary constraints on Lehmer's totient problem\\ and its companion}
\author{Deep Bhattacharjee}
\date{October 2026}

\begin{document}

\begin{abstract}
Lehmer asked in 1932 whether a composite number $n$ can satisfy
$\varphi(n)\mid n-1$, and the same question can be asked of
$\varphi(n)\mid n+1$. Neither question is answered here. What we do is
push the elementary method, the one Lehmer himself used, as far as it
will go without a computer, and record exactly where it stops. If
$3\nmid n$, a composite solution of either divisibility has at least
eleven prime factors, and eleven is the limit of the method: there is
an admissible set of eleven primes that it cannot exclude. If $3\nmid n$
and $n$ solves Lehmer's equation with at most $222$ prime factors, then
$n\equiv1\pmod 3$. If $3\mid n$, the quotient $(n-1)/\varphi(n)$ is
$\equiv1\pmod3$ and $n$ has at least $111$ prime factors. Finally
$n<(2k)^{2^{k}}/k$ when $n$ has $k$ prime factors. Every inequality comes
down to a comparison of explicit fractions.
\end{abstract}

\maketitle

\section{Introduction}

A prime $p$ satisfies $\varphi(p)=p-1$, so $\varphi(p)$ divides $p\mp1$
whenever it divides $2$. Lehmer \cite{Leh32} asked whether a composite
$n$ can satisfy
\begin{equation}\label{eq:L}
  \varphi(n)\mid n-1 ,
\end{equation}
and nobody knows. The companion divisibility
\begin{equation}\label{eq:C}
  \varphi(n)\mid n+1
\end{equation}
has the solutions $1,2,3,15,255,65535,83623935,4294967295$ and
$6992962672132095$, and it is not known whether there are others; the
eight-prime case is settled in \cite{MBB}.

Lehmer showed that a composite solution of \eqref{eq:L} is odd,
squarefree and has at least seven prime factors. Cohen and Hagis
\cite{CH80} raised seven to fourteen, Pomerance \cite{Pom77} proved
$n<k^{2^{k}}$ for $n$ with $k$ prime factors, and Burek and \.Zmija
\cite{BZ19} sharpened that bound. Hagis \cite{Hag88} showed that a
solution divisible by $3$ must have an enormous number of prime factors.
The most recent computation \cite{MBB} reaches sixteen prime factors for
\eqref{eq:L}, and sixteen for \eqref{eq:C} when $3\nmid n$.

These later bounds rest on long searches. The question behind this
note is simpler: how much can one say with nothing but the inequality
$n/\varphi(n)>2$, the coprimality of $n$ and $\varphi(n)$, and a little
arithmetic modulo $3$, if every step has to be checkable by hand? The
answer is collected in the following theorem. We write $n=p_1\cdots p_k$
with $p_1<\dots<p_k$ (Lemma~\ref{lem:basic} shows the primes are
distinct), and in the case of \eqref{eq:L} we put $a=(n-1)/\varphi(n)$.

\begin{theorem}\label{thm:main}
Let $n>3$ be composite.
\begin{enumerate}
\item[(i)] If $n$ satisfies \eqref{eq:L} or \eqref{eq:C} and $3\nmid n$,
then $k\ge11$.
\item[(ii)] If $n$ satisfies \eqref{eq:L}, $3\nmid n$ and $k\le222$, then
some prime factor of $n$ is $\equiv1\pmod3$, and hence $n\equiv1\pmod3$.
\item[(iii)] If $n$ satisfies \eqref{eq:L} and $3\mid n$, then
$a\equiv1\pmod3$, $a\ge4$ and $k\ge111$.
\item[(iv)] If $n$ satisfies \eqref{eq:L}, then $p_i\le(k-i+1)(p_1\cdots
p_{i-1}+1)$ for every $i$, so $p_1\le2k$ and $n<(2k)^{2^{k}}/k$.
\end{enumerate}
\end{theorem}

None of this is new in strength except perhaps the companion half of (i):
\cite{CH80}, \cite{Hag88}, \cite{Pom77} and \cite{MBB} prove more. The
point is that (i)--(iv) need no machine at all. Part (iv) is a slightly
weaker form of Pomerance's bound with a proof of a few lines.

Section~\ref{sec:limit} explains why the method cannot be pushed further.
The short version is that it uses the divisibility only through two of
its consequences, and an admissible set of eleven primes already
satisfies both.

\section{First reductions}

\begin{lemma}\label{lem:basic}
Let $n>3$ be composite and suppose $\varphi(n)\mid n+\varepsilon$ with
$\varepsilon=\pm1$. Then $n$ is odd and squarefree, $\gcd(n,\varphi(n))=1$,
and no prime factor of $n$ divides $q-1$ for another prime factor $q$.
The quotient $M=(n+\varepsilon)/\varphi(n)$ is at least $2$, and
\begin{equation}\label{eq:ratio}
  \frac{n}{\varphi(n)}=\prod_{i=1}^{k}\frac{p_i}{p_i-1}=M-\frac{\varepsilon}{\varphi(n)} .
\end{equation}
\end{lemma}

\begin{proof}
A common divisor of $n$ and $\varphi(n)$ divides $n+\varepsilon$ and $n$,
hence divides $1$. If $p^2\mid n$ then $p\mid\varphi(n)$, which we have
just excluded, so $n$ is squarefree, and then
$\varphi(n)=\prod(p_i-1)$, so a prime factor of $n$ dividing some $p_j-1$
would divide $\varphi(n)$. If $n$ were even it would have an odd prime
factor $q$ (it is composite and squarefree), and $2\mid q-1\mid\varphi(n)$
would contradict coprimality. Finally $M\ge1$, and $M=1$ would mean
$\varphi(n)=n+\varepsilon$; this is impossible for $\varepsilon=+1$, and
for $\varepsilon=-1$ it makes $n$ prime.
\end{proof}

For \eqref{eq:L} we have $\varepsilon=-1$ and $M=a$, so \eqref{eq:ratio}
gives $n/\varphi(n)>a\ge2$. For \eqref{eq:C}, $n/\varphi(n)=M-1/\varphi(n)
\ge2-1/\varphi(n)$, which is only a little weaker.

\begin{definition}
A set $S$ of odd primes is \emph{admissible} if $p\nmid q-1$ for all
$p,q\in S$. We write $\Pi(S)=\prod_{p\in S}p/(p-1)$.
\end{definition}

So the prime factors of a solution form an admissible set, and its
$\Pi$ is close to $2$ or above it. The only other tool we need is the
observation that $x\mapsto x/(x-1)$ decreases and exceeds $1$.

\begin{lemma}\label{lem:mono}
Let $L=\{\ell_1<\ell_2<\cdots\}$ be a set of primes and $T\subseteq L$ a
subset with $m$ elements. Then $\Pi(T)\le\prod_{j\le m}\ell_j/(\ell_j-1)$,
and the right side increases with $m$.
\end{lemma}

\begin{proof}
If $T=\{t_1<\dots<t_m\}$ then $t_j\ge\ell_j$ for each $j$. Compare
factor by factor.
\end{proof}

\section{Eleven prime factors when \texorpdfstring{$3\nmid n$}{3 does not divide n}}\label{sec:eleven}

\begin{proposition}\label{prop:eleven}
Let $S$ be an admissible set of at most ten primes, all at least $5$.
Then $\Pi(S)<1.9934$.
\end{proposition}

\begin{proof}
We sort $S$ by asking, in turn, whether it contains $5$, $7$, $13$, $17$,
$19$ and $23$, and we stop at the first prime it does not contain. This
gives seven cases, which exhaust all possibilities. In each case
admissibility removes some primes from consideration: once $5\in S$ the
primes $11,31,41,61,71$ are excluded (they are $1$ modulo $5$), once
$7\in S$ the primes $29,43,71$ are excluded, and similarly $53,79$ for
$13$, $103$ for $17$ and $47$ for $23$. Note that $11$ never needs a case
of its own: when $5\in S$ it is excluded, and when $5\notin S$ we stop
at once.

In each case the primes of $S$ that are not yet decided lie in an
explicit list, and Lemma~\ref{lem:mono} bounds their contribution by the
product over the first few entries of that list. Table~\ref{tab:cases}
shows the outcome. The second column lists the primes known to be in
$S$, the third the smallest candidates for the rest, enough of them to
bring the total to ten. The last column is the resulting product,
rounded up.
\end{proof}

The fractions in the table are in lowest terms. Anyone checking them by
hand will find it easier to multiply the factors $p/(p-1)$ one at a
time, rounding each partial product up to five decimals; the margins
are large enough for that, the smallest being about $6.7\cdot10^{-3}$ in
case~7.

\begin{proof}[Proof of Theorem~\ref{thm:main}(i)]
Suppose $3\nmid n$ and $k\le10$. All prime factors are at least $5$, and
they form an admissible set, so $n/\varphi(n)<1.9934$ by
Proposition~\ref{prop:eleven}. For \eqref{eq:L} this contradicts
$n/\varphi(n)>2$ at once. For \eqref{eq:C} it gives
$2-1/\varphi(n)<1.9934$, so $\varphi(n)<152$. But $\varphi(n)=\prod(p_i-1)$ with distinct primes
$p_i\ge5$, so $\varphi(n)\ge4\cdot6\cdot10=240$ as soon as $k\ge3$.
Hence $k=2$; and then $n/\varphi(n)\le\frac54\cdot\frac76<1.46$,
which is less than $2-1/\varphi(n)\ge2-\frac1{24}$. Either way we have a
contradiction, so $k\ge11$.
\end{proof}

\section{The residue of \texorpdfstring{$n$}{n} modulo \texorpdfstring{$3$}{3}}

When $3\nmid n$ there is a little more to be had from the prime $3$. We
need a crude but explicit bound for products over primes $\equiv2\pmod
3$.

\begin{lemma}\label{lem:sum}
For $N\ge1$,
\[
  \prod_{j=1}^{N}\frac{6j-1}{6j-2}\le\frac54\Bigl(\frac{6N+1}{7}\Bigr)^{1/6}.
\]
If $Q$ is a set of $N$ odd primes, all $\equiv2\pmod3$, then $\Pi(Q)$ is
at most the left-hand side.
\end{lemma}

\begin{proof}
For $N=1$ both sides equal $5/4$. For $j\ge2$ we use
$\log(1+x)\le x$ and then the convexity of $f(x)=1/(6x-2)$, which gives
$f(j)\le\int_{j-1/2}^{j+1/2}f$ (Hermite--Hadamard). Summing,
\[
  \sum_{j=2}^{N}\log\frac{6j-1}{6j-2}\le\int_{3/2}^{N+1/2}\frac{dx}{6x-2}
  =\frac16\log\frac{6N+1}{7},
\]
and the first claim follows. For the second, an odd prime
$\equiv2\pmod3$ is $\equiv5\pmod6$, so the $j$-th smallest element of $Q$
is at least $6j-1$, and we argue as in Lemma~\ref{lem:mono}.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:main}(ii)]
Let $n$ satisfy \eqref{eq:L} with $3\nmid n$. If some prime factor $p$ is
$\equiv1\pmod3$ then $3\mid p-1\mid\varphi(n)\mid n-1$. Suppose instead
that every $p_i$ is $\equiv2\pmod3$. Then each $p_i-1\equiv1$, so
$\varphi(n)\equiv1$, and $n\equiv(-1)^{k}\pmod3$. Reducing
$n-1=a\varphi(n)$ modulo $3$ gives $a\equiv(-1)^{k}-1$, which is $0$ or
$1$ modulo $3$, never $2$. So $a\ne2$, and as $a\ge2$ we get $a\ge3$. By
\eqref{eq:ratio} and Lemma~\ref{lem:sum},
\[
  3<\frac54\Bigl(\frac{6k+1}{7}\Bigr)^{1/6},\qquad\text{that is,}\qquad
  \frac{6k+1}{7}>\Bigl(\frac{12}{5}\Bigr)^{6}=\frac{2985984}{15625}>191.1 .
\]
For $k\le222$ the left side is at most $1333/7<190.5$. Hence $k\ge223$.
\end{proof}

\section{Solutions divisible by \texorpdfstring{$3$}{3}}

For the companion equation nothing of this kind is possible: $15$,
$255$ and the other known solutions are divisible by $3$. For Lehmer's
equation the prime $3$ is very costly.

\begin{proof}[Proof of Theorem~\ref{thm:main}(iii)]
Let $n$ satisfy \eqref{eq:L} with $3\mid n$. No other prime factor $q$
can be $\equiv1\pmod3$, so each is $\equiv2$ and $q-1\equiv1\pmod3$.
Hence $\varphi(n)=2\prod_{q\ne3}(q-1)\equiv2$, while $n-1\equiv2\pmod 3$.
From $n-1=a\varphi(n)$ we get $2a\equiv2$, so $a\equiv1\pmod3$, and
$a\ge2$ forces $a\ge4$. Write $Q$ for the set of the other $N=k-1$ prime
factors. Then $\frac32\Pi(Q)>4$, and Lemma~\ref{lem:sum} gives
\[
  \frac83<\frac54\Bigl(\frac{6N+1}{7}\Bigr)^{1/6},\qquad\text{that is,}\qquad
  \frac{6N+1}{7}>\Bigl(\frac{32}{15}\Bigr)^{6}=\frac{1073741824}{11390625}>94.26 .
\]
For $N\le109$ the left side is at most $655/7<93.58$, so $N\ge110$ and
$k\ge111$.
\end{proof}

The true threshold for this argument is a little higher, since the
estimate in Lemma~\ref{lem:sum} gives away a constant factor; Hagis
\cite{Hag88} obtains far more by using much more.

\section{An upper bound}

The lower bounds above say nothing about size. The following argument,
a variant of Pomerance's, bounds each prime factor in terms of the ones
before it.

\begin{lemma}\label{lem:prefix}
Let $n=p_1\cdots p_k$ satisfy \eqref{eq:L}, let $1\le i\le k$, and put
$m=p_1\cdots p_{i-1}$ (so $m=1$ when $i=1$) and $r=n/m$. Then
$m/\varphi(m)<a$, and consequently $r/\varphi(r)>1+1/m$.
\end{lemma}

\begin{proof}
Suppose $m/\varphi(m)\ge a$. Every prime factor of $r$ is at most $p_k$,
so $r/\varphi(r)\ge1+1/(p_k-1)$, and
\[
  \frac1{\varphi(n)}=\frac{n}{\varphi(n)}-a\ge a\Bigl(\frac{r}{\varphi(r)}-1\Bigr)
  \ge\frac{2}{p_k-1}.
\]
That says $2\varphi(n)\le p_k-1$, which is absurd because $p_k-1$
divides $\varphi(n)$. So $m<a\varphi(m)$, and since these are integers,
$a\varphi(m)\ge m+1$. Dividing \eqref{eq:ratio} by $m/\varphi(m)$ gives
$r/\varphi(r)>a\varphi(m)/m\ge1+1/m$.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:main}(iv)]
Let $s=k-i+1$ be the number of prime factors of $r$, each at least
$p_i$. Using $1+x\le e^{x}$ and $\log(1+1/m)\ge1/(m+1)$,
\[
  e^{1/(m+1)}\le1+\frac1m<\frac{r}{\varphi(r)}\le\Bigl(1+\frac1{p_i-1}\Bigr)^{s}
  \le e^{s/(p_i-1)},
\]
so $p_i-1<s(m+1)$, and $p_i\le s(m+1)$ because $p_i$ is an integer. For
$i=1$ this is $p_1\le2k$. Now let $x_i=p_1\cdots p_i$, so $x_0=1$ and
$x_k=n$. We have $x_i\le k\,x_{i-1}(x_{i-1}+1)$, hence
$x_i+1\le k(x_{i-1}+1)^2$, and with $y_i=k(x_i+1)$ this reads
$y_i\le y_{i-1}^2$. So $y_k\le y_0^{2^{k}}=(2k)^{2^{k}}$, which is the
claim.
\end{proof}

In particular there are only finitely many solutions of \eqref{eq:L}
with a given number of prime factors, and in principle each case can be
decided. In practice the bound is far too large to search, and that is
exactly why \cite{CH80} and \cite{MBB} use cleverer searches.

\section{Where the method stops}\label{sec:limit}

Everything in Section~\ref{sec:eleven} uses only two facts about $n$:
its prime factors form an admissible set, and their $\Pi$ is at least
about $2$. Eleven primes are enough to satisfy both.

\begin{proposition}\label{prop:limit}
The set $S=\{5,7,13,17,19,23,37,59,67,73,83\}$ is admissible, and
\[
  \Pi(S)=\frac{2995513256722805}{1484664878923776}>2 .
\]
\end{proposition}

\begin{proof}
The numbers $p-1$ for $p\in S$ are $4,6,12,16,18,22,36,58,66,72,82$.
Their odd prime factors are $3,11,29,41$, none of which lies in $S$. For
the inequality, $2\cdot1484664878923776=2969329757847552$, which is
smaller than the numerator.
\end{proof}

Its product
$n'=5\cdot7\cdots83$ is of course not a solution: $n'-1$ is not a
multiple of $\varphi(n')$. To go further one has to use the divisibility
itself and not only its consequences, and that is where every known
argument turns into a search. The obstacles are of three kinds.
\begin{enumerate}
\item[(a)] Nothing in the method gets worse as $k$ grows. We do not
prove that admissible sets with $\Pi(S)>2$ exist for every $k\ge11$,
but we see no reason why they should not, and the product over all
primes diverges.
\item[(b)] The upper bound of Theorem~\ref{thm:main}(iv) grows doubly
exponentially with $k$, while the lower bounds only constrain $k$. They
leave a finite but enormous range for each $k$, and they never meet.
\item[(c)] For the companion equation with $3\mid n$ there are genuine
solutions, so no argument of this kind can say anything there; the
residual question is whether the nine known solutions are all.
\end{enumerate}
Both of Lehmer's questions therefore remain open, and this note claims
nothing beyond Theorem~\ref{thm:main}.
